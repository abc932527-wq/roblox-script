-- REMOTE CACHE
-- ==========================================
local Remotes = {}
local RemoteCache = {}

local function FindRemotes()
    local patterns = {
        Roll = {"roll", "dice", "summon"},
        Rebirth = {"rebirth", "prestige"},
        Equip = {"equip", "set"},
        Upgrade = {"upgrade", "enhance"},
        Claim = {"claim", "reward"},
        Reroll = {"reroll", "reset"},
        Sell = {"sell", "discard"},
        Tower = {"tower", "start"},
        Shop = {"shop", "buy"}
    }

    for _, obj in ipairs(ReplicatedStorage:GetDescendants()) do
        if obj:IsA("RemoteEvent") or obj:IsA("RemoteFunction") then
            local name = obj.Name:lower()
            for category, keywords in pairs(patterns) do
                for _, keyword in ipairs(keywords) do
                    if name:find(keyword) and not Remotes[category] then
                        Remotes[category] = obj
                        break
                    end
                end
            end
        end
    end
end

FindRemotes()

-- ==========================================
-- SAFE REMOTE FIRE (Anti-Spam)
-- ==========================================
local lastFire = {}

function SafeFireRemote(category, ...)
    local remote = Remotes[category]
    if not remote then
        return false
    end

    local now = tick()
    local lastTime = lastFire[remote] or 0

    if (now - lastTime) < 0.6 then
        return false
    end

    lastFire[remote] = now

    local success, err = pcall(function()
        if remote:IsA("RemoteEvent") then
            remote:FireServer(...)
        else
            remote:InvokeServer(...)
        end
    end)

    return success
end

-- ==========================================
-- MAIN LOOP (Optimized)
-- ==========================================
local timers = {
    main = 0,
    equip = 0,
    roll = 0
}

RunService.Heartbeat:Connect(function(deltaTime)
    -- Update timers
    timers.main = timers.main + deltaTime
    timers.equip = timers.equip + deltaTime
    timers.roll = timers.roll + deltaTime

    -- Anti AFK (lightweight)
    if S.AntiAFK then
        pcall(function()
            local vUser = game:GetService("VirtualUser")
            vUser:CaptureController()
            vUser:ClickButton2(Vector2.new())
        end)
    end

    -- Main logic throttle (every 1 second)
    if timers.main < 1.0 then
        return
    end
    timers.main = 0

    -- Auto Roll (throttled separately)
    if S.AutoRoll and timers.roll >= 2.0 then
        timers.roll = 0
        SafeFireRemote("Roll")

        if S.SkipAnim then
            local playerGui = LP:FindFirstChild("PlayerGui")
            if playerGui then
                for _, gui in ipairs(playerGui:GetChildren()) do
                    if gui.Name:lower():find("roll") or gui.Name:lower():find("summon") then
                        local skipBtn = gui:FindFirstChild("SkipButton", true)
                        if skipBtn and skipBtn:IsA("TextButton") then
                            pcall(function() skipBtn:Fire() end)
                        end
                    end
                end
            end
        end
    end

    -- Auto Rebirth
    if S.AutoRebirth then
        SafeFireRemote("Rebirth")
    end

    -- Auto Collect
    if S.AutoCollect then
        SafeFireRemote("Claim", "Coins")
        SafeFireRemote("Claim", "Gems")
    end

    -- Auto Claim Rewards
    if S.AutoClaim then
        SafeFireRemote("Claim", "Daily")
        SafeFireRemote("Claim", "Offline")
    end

    -- Auto Upgrade Units
    if S.AutoUpgradeUnits then
        SafeFireRemote("Upgrade", "Unit")
    end

    -- Auto Upgrade Skills
    if S.AutoUpgradeSkills then
        SafeFireRemote("Upgrade", "Skill")
    end

    -- Auto Dice Shop
    if S.AutoDiceShop then
        SafeFireRemote("Shop", "Buy")
    end

    -- Auto Trait Reroll
    if S.AutoTraitReroll then
        local targetUnit = S.TargetUnit or ""
        local targetTrait = S.TargetTrait or "Godly"
        SafeFireRemote("Reroll", "Trait", targetUnit, targetTrait)
    end

    -- Auto Grade Reroll
    if S.AutoGradeReroll then
        local targetUnit = S.TargetUnit or ""
        local targetGrade = S.TargetGrade or "SSS"
        SafeFireRemote("Reroll", "Grade", targetUnit, targetGrade)
    end

    -- Auto Tower
    if S.AutoTower then
        local towerName = S.TargetTower or "Tower 1"
        SafeFireRemote("Tower", "Start", towerName)
    end

    -- Auto Equip (interval based)
    if timers.equip >= (S.EquipInterval or 15) then
        timers.equip = 0

        if S.AutoEquipBest then
            SafeFireRemote("Equip", "BestUnit")
        end

        if S.AutoEquipBestDice then
            SafeFireRemote("Equip", "BestDice")
        end

        if S.AutoEquipTeam then
            SafeFireRemote("Equip", "BestTeam")
        end
    end
end)
