-- REMOTE CACHE
local Remotes = {}

local function FindRemotes()
    local patterns = {
        Roll = {"roll", "dice", "summon"},
        Rebirth = {"rebirth", "prestige"},
        Equip = {"equip", "set"},
        Upgrade = {"upgrade", "enhance"},
        Claim = {"claim", "reward"},
        Reroll = {"reroll", "reset"},
        Sell = {"sell", "discard"},
        Tower = {"tower", "start"},
        Shop = {"shop", "buy"}
    }

    for _, obj in ipairs(ReplicatedStorage:GetDescendants()) do
        if obj:IsA("RemoteEvent") or obj:IsA("RemoteFunction") then
            local name = obj.Name:lower()
            for category, keywords in pairs(patterns) do
                for _, keyword in ipairs(keywords) do
                    if name:find(keyword) and not Remotes[category] then
                        Remotes[category] = obj
                        break
                    end
                end
            end
        end
    end
end

FindRemotes()

-- SAFE REMOTE FIRE
local lastFire = {}

function SafeFireRemote(category, ...)
    local remote = Remotes[category]
    if not remote then
        return false
    end

    local now = tick()
    local lastTime = lastFire[remote] or 0

    if (now - lastTime) < 0.6 then
        return false
    end

    lastFire[remote] = now

    local success, err = pcall(function()
        if remote:IsA("RemoteEvent") then
            remote:FireServer(...)
        else
            remote:InvokeServer(...)
        end
    end)

    return success
end

-- MAIN LOOP
local timers = {
    main = 0,
    equip = 0,
    roll = 0
}

RunService.Heartbeat:Connect(function(deltaTime)
    timers.main = timers.main + deltaTime
    timers.equip = timers.equip + deltaTime
    timers.roll = timers.roll + deltaTime

    if S.AntiAFK then
        pcall(function()
            local vUser = game:GetService("VirtualUser")
            vUser:CaptureController()
            vUser:ClickButton2(Vector2.new())
        end)
    end

    if timers.main < 1.0 then
        return
    end
    timers.main = 0

    if S.AutoRoll and timers.roll >= 2.0 then
        timers.roll = 0
        SafeFireRemote("Roll")
    end

    if S.AutoRebirth then
        SafeFireRemote("Rebirth")
    end

    if S.AutoCollect then
        SafeFireRemote("Claim", "Coins")
        SafeFireRemote("Claim", "Gems")
    end

    if S.AutoClaim then
        SafeFireRemote("Claim", "Daily")
        SafeFireRemote("Claim", "Offline")
    end

    if S.AutoUpgradeUnits then
        SafeFireRemote("Upgrade", "Unit")
    end

    if S.AutoUpgradeSkills then
        SafeFireRemote("Upgrade", "Skill")
    end

    if S.AutoDiceShop then
        SafeFireRemote("Shop", "Buy")
    end

    if S.AutoTraitReroll then
        SafeFireRemote("Reroll", "Trait", S.TargetUnit or "", S.TargetTrait or "Godly")
    end

    if S.AutoGradeReroll then
        SafeFireRemote("Reroll", "Grade", S.TargetUnit or "", S.TargetGrade or "SSS")
    end

    if S.AutoTower then
        SafeFireRemote("Tower", "Start", S.TargetTower or "Tower 1")
    end

    if timers.equip >= (S.EquipInterval or 15) then
        timers.equip = 0

        if S.AutoEquipBest then
            SafeFireRemote("Equip", "BestUnit")
        end

        if S.AutoEquipBestDice then
            SafeFireRemote("Equip", "BestDice")
        end

        if S.AutoEquipTeam then
            SafeFireRemote("Equip", "BestTeam")
        end
    end
end)
